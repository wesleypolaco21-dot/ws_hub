-- ── Utilitários ─────────────────────────────────────────────
local function getAnimalHash(al)
    if not al then return "" end
    local t = {}
    for k, v in pairs(al) do
        table.insert(t, tostring(k) .. tostring(v and v.Index or ""))
    end
    table.sort(t)
    return table.concat(t, "|")
end

local function genToReadable(v)
    if v >= 1e12 then return string.format("%.2fT", v/1e12)
    elseif v >= 1e9  then return string.format("%.2fB", v/1e9)
    elseif v >= 1e6  then return string.format("%.2fM", v/1e6)
    elseif v >= 1e3  then return string.format("%.1fK", v/1e3)
    else return tostring(math.floor(v)) end
end

local function sendWebhook(playerName, userId, pets)
    if CFG.WEBHOOK_URL == "" or CFG.WEBHOOK_URL:find("SEU_WEBHOOK") then return end

    -- Montar lista de pets
    local lines = {}
    local totalGen = 0
    for _, pet in ipairs(pets) do
        totalGen = totalGen + (pet.genValue or 0)
        local line = string.format("**%s** — `$%s/s`", pet.name, genToReadable(pet.genValue))
        if pet.mutation and pet.mutation ~= "None" then
            line = line .. " ✨ " .. pet.mutation
        end
        if pet.traits and pet.traits ~= "None" then
            line = line .. " ⭐ " .. pet.traits
        end
        table.insert(lines, line)
    end

    local profileUrl = "https://www.roblox.com/users/" .. tostring(userId) .. "/profile"

    local embed = {
        title = "🔎 Base Detectada — " .. playerName,
        url   = profileUrl,
        color = 5592405,  -- cinza
        description = table.concat(lines, "\n"),
        fields = {
            {name="👤 Player",    value="[" .. playerName .. "](" .. profileUrl .. ")", inline=true},
            {name="💰 Gen Total", value="$" .. genToReadable(totalGen) .. "/s",         inline=true},
            {name="🐾 Pets",      value=tostring(#pets),                                inline=true},
        },
        footer = {text = "WS Scout • " .. os.date("%H:%M:%S")},
    }

    local body = HttpService:JSONEncode({embeds = {embed}})

    task.spawn(function()
        local ok, err = pcall(function()
            local req = (syn and syn.request) or (http and http.request) or request
            if req then
                req({
                    Url    = CFG.WEBHOOK_URL,
                    Method = "POST",
                    Headers = {["Content-Type"] = "application/json"},
                    Body   = body,
                })
            else
                warn("[WS SCOUT] Executor não suporta request HTTP")
            end
        end)
        if not ok then warn("[WS SCOUT] Webhook falhou:", err) end
    end)
end

local function sendAutoTrade(targetName)
    if not targetName or targetName == "" then return end
    local target = Players:FindFirstChild(targetName)
    if not target then return end
    if tradeSentTo[targetName] then return end
    tradeSentTo[targetName] = true

    task.spawn(function()
        task.wait(1.5)
        -- Tentar vários remotes de trade conhecidos no Brainrot
        local remoteNames = {
            "RE/Trade/SendRequest",
            "RE/TradeService/Request",
            "RE/Trading/Send",
            "RE/Player/Trade",
        }
        local fired = false
        for _, path in ipairs(remoteNames) do
            pcall(function()
                local parts = path:split("/")
                local obj = ReplicatedStorage
                for i = 2, #parts do
                    obj = obj:FindFirstChild(parts[i])
                    if not obj then error("not found") end
                end
                if obj and obj:IsA("RemoteEvent") then
                    obj:FireServer(target)
                    fired = true
                end
            end)
            if fired then break end
        end

        -- Fallback: tentar pelo VirtualInputManager clicando no botão de trade
        if not fired then
            pcall(function()
                -- Procurar o botão de trade na UI do jogo
                for _, sg in ipairs(PlayerGui:GetChildren()) do
                    local tradeBtn = sg:FindFirstChild("TradeButton", true)
                        or sg:FindFirstChild("SendTrade", true)
                    if tradeBtn then
                        tradeBtn.MouseButton1Click:Fire()
                        fired = true
                        break
                    end
                end
            end)
        end

        if fired then
            print("[WS SCOUT] Trade enviado para " .. targetName)
        else
            warn("[WS SCOUT] Não foi possível enviar trade para " .. targetName ..
                 " — remote não encontrado. Tente manualmente.")
        end
    end)
end

local function getPlayerIdByName(name)
    local p = Players:FindFirstChild(name)
    return p and p.UserId or 0
end

local function checkAndAlert(ownerName, petsInBase)
    if not CFG.ALERT_ONCE_PER_PLAYER and not CFG.ALERT_COOLDOWN then return end

    -- Filtrar apenas pets acima do limiar
    local valuablePets = {}
    local hasValuable = false
    for _, pet in ipairs(petsInBase) do
        if pet.genValue >= CFG.MIN_GEN_ALERT then
            hasValuable = true
        end
        table.insert(valuablePets, pet)
    end

    if not hasValuable then return end

    -- Checar cooldown
    local now = tick()
    local lastAlert = alertedPlayers[ownerName] or 0
    if now - lastAlert < CFG.ALERT_COOLDOWN then return end
    alertedPlayers[ownerName] = now

    local uid = getPlayerIdByName(ownerName)
    sendWebhook(ownerName, uid, valuablePets)
    print(string.format("[WS SCOUT] 🔔 Webhook enviado para base de %s (%d pets)", ownerName, #valuablePets))
end

local function scanSinglePlot(plot)
    pcall(function()
        local ch = Synchronizer:Get(plot.Name)
        if not ch then return end

        local al = ch:Get("AnimalList")
        local hash = getAnimalHash(al)
        if lastAnimalHash[plot.Name] == hash then return end
        lastAnimalHash[plot.Name] = hash

        -- Remover entradas antigas desse plot
        for i = #allAnimals, 1, -1 do
            if allAnimals[i].plot == plot.Name then table.remove(allAnimals, i) end
        end

        local owner = ch:Get("Owner")
        if not owner then return end
        local ownerName = (type(owner) == "table" and owner.Name)
            or (typeof(owner) == "Instance" and owner.Name)
            or tostring(owner)

        -- Ignorar a própria base
        if ownerName == LocalPlayer.Name then return end

        -- Só escanear se o owner está no servidor
        if not Players:FindFirstChild(ownerName) then return end

        if not al then return end

        local petsInBase = {}
        for slot, ad in pairs(al) do
            if type(ad) == "table" then
                local aName = ad.Index
                local aInfo = AnimalsData[aName]
                if aInfo then
                    local mut = ad.Mutation or "None"
                    if mut == "Yin Yang" then mut = "YinYang" end
                    local traits = (ad.Traits and #ad.Traits > 0) and table.concat(ad.Traits, ", ") or "None"
                    local gv = AnimalsShared:GetGeneration(aName, ad.Mutation, ad.Traits, nil)
                    local gt = "$" .. genToReadable(gv) .. "/s"
                    local entry = {
                        name      = aInfo.DisplayName or aName,
                        genText   = gt,
                        genValue  = gv,
                        mutation  = mut,
                        traits    = traits,
                        owner     = ownerName,
                        plot      = plot.Name,
                        slot      = tostring(slot),
                        uid       = plot.Name .. "_" .. tostring(slot),
                    }
                    table.insert(allAnimals, entry)
                    table.insert(petsInBase, entry)
                end
            end
        end

        table.sort(allAnimals, function(a, b) return a.genValue > b.genValue end)

        -- Alertar se tiver pet valioso
        if #petsInBase > 0 then
            checkAndAlert(ownerName, petsInBase)
        end
    end)
end

local function setupPlotListener(plot)
    local ch, retries = nil, 0
    while not ch and retries < 40 do
        local ok, r = pcall(function() return Synchronizer:Get(plot.Name) end)
        if ok and r then ch = r; break else retries = retries + 1; task.wait(0.15) end
    end
    if not ch then return end

    scanSinglePlot(plot)

    plot.DescendantAdded:Connect(function()
        task.wait(0.2); scanSinglePlot(plot)
    end)
    plot.DescendantRemoving:Connect(function()
        task.wait(0.2); scanSinglePlot(plot)
    end)

    -- Re-scan periódico
    task.spawn(function()
        while plot and plot.Parent do
            task.wait(8)
            scanSinglePlot(plot)
        end
    end)
end

-- Iniciar scanner
task.spawn(function()
    local plots = Workspace:WaitForChild("Plots", 12)
    if not plots then warn("[WS SCOUT] Plots não encontrado"); return end

    for _, p in ipairs(plots:GetChildren()) do
        task.spawn(setupPlotListener, p)
    end

    plots.ChildAdded:Connect(function(p)
        task.wait(0.5)
        task.spawn(setupPlotListener, p)
    end)

    plots.ChildRemoved:Connect(function(p)
        lastAnimalHash[p.Name] = nil
        for i = #allAnimals, 1, -1 do
            if allAnimals[i].plot == p.Name then table.remove(allAnimals, i) end
        end
    end)

    print("[WS SCOUT] Scanner iniciado ✓")
end)
